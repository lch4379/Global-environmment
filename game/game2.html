// 게임 설정 const GAME_TIME = 30; // 게임 시간(초) const MOLE_INTERVAL = 650; // 두더지가 바뀌는 간격(ms) // HTML 요소 가져오기 const holes = document.querySelectorAll(".hole"); const scoreDisplay = document.getElementById("score"); const timeDisplay = document.getElementById("time"); const bestDisplay = document.getElementById("best"); const message = document.getElementById("message"); const startButton = document.getElementById("startButton"); // 게임 상태 let score = 0; let timeLeft = GAME_TIME; let currentMole = -1; let playing = false; let timer = null; let moleTimer = null; // 최고 점수 불러오기 let bestScore = 0; try { bestScore = Number(localStorage.getItem("moleBestScore")) || 0; } catch (error) { bestScore = 0; } bestDisplay.textContent = bestScore; // 모든 구멍을 비우기 function clearHoles() { holes.forEach((hole) => { hole.textContent = "🕳️"; hole.classList.remove("mole"); }); currentMole = -1; } // 두더지를 랜덤한 구멍에 나타내기 function showMole() { if (!playing) return; clearHoles(); // 직전에 나온 구멍과 다른 곳을 선택 let nextMole; do { nextMole = Math.floor(Math.random() * holes.length); } while (holes.length > 1 && nextMole === currentMole); currentMole = nextMole; holes[currentMole].textContent = "🐹"; holes[currentMole].classList.add("mole"); } // 게임 시작 function startGame() { // 기존 타이머 정리 clearInterval(timer); clearInterval(moleTimer); score = 0; timeLeft = GAME_TIME; playing = true; scoreDisplay.textContent = score; timeDisplay.textContent = timeLeft; message.textContent = "두더지를 잡아 봐!"; startButton.textContent = "🔄 다시 시작"; clearHoles(); showMole(); // 일정 시간마다 두더지 위치 변경 moleTimer = setInterval(showMole, MOLE_INTERVAL); // 1초마다 남은 시간 감소 timer = setInterval(() => { timeLeft--; timeDisplay.textContent = timeLeft; if (timeLeft bestScore) { bestScore = score; bestDisplay.textContent = bestScore; try { localStorage.setItem("moleBestScore", String(bestScore)); message.textContent = `🎉 신기록 달성! ${score}점!`; } catch (error) { message.textContent = `게임 종료! ${score}점!`; } } else { message.textContent = `게임 종료! 최종 점수는 ${score}점이야!`; } startButton.textContent = "🎮 다시 도전"; } // 구멍 터치 처리 holes.forEach((hole, index) => { hole.addEventListener("click", () => { if (!playing) return; if (index === currentMole) { // 두더지를 정확히 터치했을 때 score++; scoreDisplay.textContent = score; // 같은 두더지를 연속으로 누르지 못하도록 숨기기 hole.textContent = "🕳️"; hole.classList.remove("mole"); currentMole = -1; message.textContent = "잘했어! +1점 🐹"; // 다음 두더지를 즉시 등장시킴 clearInterval(moleTimer); showMole(); moleTimer = setInterval(showMole, MOLE_INTERVAL); } }); }); // 시작 버튼 startButton.addEventListener("click", startGame);
